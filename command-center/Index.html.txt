function finRows(){
  const rows = [];
  const today = new Date();
  (D.finance||[]).forEach(f => {
    const base = {src:"Ledger", row:f._row, type:/expense/i.test(f.Type)?"Expense":"Income", cat:f.Category||"", desc:f.Description||"", amount:Math.abs(num(f.Amount)), client:f.Client||"", method:f.Method||"", prod:f.Product||"", repeats:f.Repeats||"", raw:num(f.Amount)};
    const d0 = new Date(f.Date); if (isNaN(d0)) return;
    if (!base.repeats) { rows.push(Object.assign({date:ymd(d0)}, base)); return; }
    const stepM = base.repeats === "Yearly" ? 12 : 1;
    for (let d = new Date(d0), n = 0; d <= today && n < 120; n++) { rows.push(Object.assign({date:ymd(d), recurring:n>0}, base)); d = new Date(d.getFullYear(), d.getMonth()+stepM, d0.getDate()); }
  });
  (D.orders||[]).filter(o => o.Status === "Paid").forEach(o => rows.push({src:"Order", date:ymd(o.Received), type:"Income", cat:"Sales", desc:o.Item||"Order", amount:num(o["Amount due"]), client:o.Business||o.Name||"", method:o.Method||"", prod:(products(o.Item)[0]||o.Item||"").replace(/:.*$/,"")}));
  return rows;
}
