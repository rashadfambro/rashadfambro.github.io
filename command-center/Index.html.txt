document.addEventListener("click", e => {
  const t = e.target.closest("[data-tab]"); if (t) { tab = t.dataset.tab; render(); return; }
  const a = e.target.closest("[data-act]");
  if (a) { const labels={record_payment:"Record a payment and email a renewed license link",resend:"Email the client their link again",pause:"Pause this license (client is emailed)",cancel:"Cancel this license (client is emailed)",mark_founding:"Mark as a founding client"};
    if (!confirm(labels[a.dataset.act] + "?")) return;
    a.disabled = true; call("licenseAction", a.dataset.act, a.dataset.key).then(() => { toast("Done"); load(); }).catch(err => { a.disabled=false; toast("Failed: " + err); }); }
});
$("#refresh").addEventListener("click", load);
document.addEventListener("change", e => { const s = e.target.closest("[data-lead]"); if (!s) return;
  call("setLeadStatus", Number(s.dataset.lead), s.value).then(() => toast("Status saved")).catch(err => toast("Failed: " + err)); });
document.addEventListener("submit", e => { e.preventDefault(); const f = e.target, v = Object.fromEntries(new FormData(f));
  const b = f.querySelector("button[type=submit]"); b.disabled = true;
  const p = f.id === "reqForm" ? call("addRequest", v.agent, v.priority, v.topic, v.text) : call("addFinance", v);
  p.then(() => { toast("Saved"); f.reset(); load(); }).catch(err => toast("Failed: " + err)).finally(() => b.disabled = false); });
